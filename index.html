คำสั่ง SQL สร้างตาราง & Storage Bucket (นำไปรันใน Supabase > SQL Editor): 
-- 1. สร้างตารางเก็บข้อมูลคลิป
CREATE TABLE IF NOT EXISTS public.recaps (
    id TEXT PRIMARY KEY,
    title TEXT NOT NULL,
    titles JSONB DEFAULT '[]'::jsonb,
    description TEXT,
    video_url TEXT,
    thumbnail_url TEXT,
    folder_name TEXT,
    series TEXT,
    duration TEXT,
    tags TEXT[] DEFAULT '{}',
    created_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE public.recaps ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public read" ON public.recaps FOR SELECT USING (true);
CREATE POLICY "Allow public insert" ON public.recaps FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow public update" ON public.recaps FOR UPDATE USING (true);

-- 2. สร้าง Storage Bucket สำหรับเก็บไฟล์วิดีโอ .mp4 และปก .jpg
INSERT INTO storage.buckets (id, name, public) VALUES ('recaps', 'recaps', true)
ON CONFLICT (id) DO UPDATE SET public = true;

CREATE POLICY "Public Read Recaps" ON storage.objects FOR SELECT USING (bucket_id = 'recaps');
CREATE POLICY "Public Insert Recaps" ON storage.objects FOR INSERT WITH CHECK (bucket_id = 'recaps');
